# Standalone CMake for Cricket Simulation only (no SDL2/OpenGL needed)
# Usage: cmake -B build_sim -S . -DCMAKE_TOOLCHAIN_FILE=CMakeLists_sim.txt
# Or simply: g++ -std=c++17 -O2 -Isrc src/cricket/*.cpp -o CricketSim

cmake_minimum_required(VERSION 3.15)
project(CricketSim VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

set(SOURCES
    src/cricket/cricket_main.cpp
    src/cricket/MatchEngine.cpp
    src/cricket/CricketTypes.cpp
    src/cricket/TeamData.cpp
)

add_executable(CricketSim ${SOURCES})
target_include_directories(CricketSim PRIVATE ${CMAKE_SOURCE_DIR}/src)

if(MSVC)
    target_compile_options(CricketSim PRIVATE /W4 /O2)
else()
    target_compile_options(CricketSim PRIVATE -Wall -Wextra -O2)
endif()
